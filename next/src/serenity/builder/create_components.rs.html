7/// A builder for creating a components action row in a message.
8///
9/// [Discord docs](https://discord.com/developers/docs/interactions/message-components#component-object).
10#[derive(Clone, Debug)]
11#[must_use]
12pub enum CreateActionRow<'a> {
13    Buttons(Cow<'a, [CreateButton<'a>]>),
14    SelectMenu(CreateSelectMenu<'a>),
15    /// Only valid in modals!
16    InputText(CreateInputText<'a>),
17}
18
19impl<'a> CreateActionRow<'a> {
20    pub fn buttons(buttons: impl Into<Cow<'a, [CreateButton<'a>]>>) -> Self {
21        Self::Buttons(buttons.into())
22    }
23
24    pub fn select_menu(select_menu: impl Into<CreateSelectMenu<'a>>) -> Self {
25        Self::SelectMenu(select_menu.into())
26    }
27
28    pub fn input_text(input_text: impl Into<CreateInputText<'a>>) -> Self {
29        Self::InputText(input_text.into())
30    }
31}
32
33impl serde::Serialize for CreateActionRow<'_> {
34    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
35        use serde::ser::SerializeMap as _;
36
37        let mut map = serializer.serialize_map(Some(2))?;
38        map.serialize_entry("type", &1_u8)?;
39
40        match self {
41            CreateActionRow::Buttons(buttons) => map.serialize_entry("components", &buttons)?,
42            CreateActionRow::SelectMenu(select) => map.serialize_entry("components", &[select])?,
43            CreateActionRow::InputText(input) => map.serialize_entry("components", &[input])?,
44        }
45
46        map.end()
47    }
48}

93        Self {
94            style: ButtonStyle::Unknown(6),
95            kind: ComponentType::Button,
96            url: None,
97            custom_id: None,
98            emoji: None,
99            label: None,
100            sku_id: Some(sku_id.into()),

162impl From<Button> for CreateButton<'_> {
163    fn from(button: Button) -> Self {
164        let (style, url, custom_id, sku_id) = match button.data {
165            ButtonKind::Link {
166                url,
167            } => (ButtonStyle::Unknown(5), Some(url.into()), None, None),
168            ButtonKind::Premium {
169                sku_id,
170            } => (ButtonStyle::Unknown(6), None, None, Some(sku_id)),
171            ButtonKind::NonLink {
172                custom_id,
173                style,
174            } => (style, None, Some(custom_id.into()), None),
175        };
176
177        Self {
178            style,
179            kind: ComponentType::Button,
180            url,
181            custom_id,
182            sku_id,
183            label: button.label.map(Into::into),
184            emoji: button.emoji,
185            disabled: button.disabled,
186        }
187    }
188}

195

202        let mut map = serializer.serialize_map(Some(2))?;
203        map.serialize_entry("id", &id)?;
204        map.serialize_entry("type", kind)?;
205        map.end()
206    }

209/// [Discord docs](https://discord.com/developers/docs/interactions/message-components#select-menu-object-select-menu-structure).
210#[derive(Clone, Debug)]
211pub enum CreateSelectMenuKind<'a> {
212    String {
213        options: Cow<'a, [CreateSelectMenuOption<'a>]>,
214    },
215    User {
216        default_users: Option<Cow<'a, [UserId]>>,
217    },
218    Role {
219        default_roles: Option<Cow<'a, [RoleId]>>,
220    },
221    Mentionable {
222        default_users: Option<Cow<'a, [UserId]>>,
223        default_roles: Option<Cow<'a, [RoleId]>>,
224    },
225    Channel {
226        channel_types: Option<Cow<'a, [ChannelType]>>,
227        default_channels: Option<Cow<'a, [ChannelId]>>,
228    },
229}
230
231impl Serialize for CreateSelectMenuKind<'_> {
232    fn serialize<S: serde::Serializer>(&self, serializer: S) -> Result<S::Ok, S::Error> {
233        #[derive(Serialize)]
234        struct Json<'a> {
235            #[serde(rename = "type")]
236            kind: u8,
237            #[serde(skip_serializing_if = "Option::is_none")]
238            options: Option<&'a [CreateSelectMenuOption<'a>]>,
239            #[serde(skip_serializing_if = "Option::is_none")]
240            channel_types: Option<&'a [ChannelType]>,
241            #[serde(skip_serializing_if = "<[_]>::is_empty")]
242            default_values: &'a [CreateSelectMenuDefault],
243        }
244
245        #[expect(clippy::ref_option)]
246        fn map<'a>(
247            values: &'a Option<Cow<'a, [impl Into<Mention> + Copy]>>,
248        ) -> impl Iterator<Item = CreateSelectMenuDefault> + 'a {
249            // Calling `.iter().flatten()` on the `Option` treats `None` like an empty vec
250            values
251                .as_ref()
252                .map(|s| s.iter())
253                .into_iter()
254                .flatten()
255                .map(|&i| CreateSelectMenuDefault(i.into()))
256        }
257
258        #[rustfmt::skip]
259        let default_values = match self {
260            Self::String { .. } => vec![],
261            Self::User { default_users: default_values } => map(default_values).collect(),
262            Self::Role { default_roles: default_values } => map(default_values).collect(),
263            Self::Mentionable { default_users, default_roles } => {
264                let users = map(default_users);
265                let roles = map(default_roles);
266                users.chain(roles).collect()
267            },
268            Self::Channel { channel_types: _, default_channels: default_values } => map(default_values).collect(),
269        };
270
271        #[rustfmt::skip]
272        let json = Json {
273            kind: match self {
274                Self::String { .. } => 3,
275                Self::User { .. } => 5,
276                Self::Role { .. } => 6,
277                Self::Mentionable { .. } => 7,
278                Self::Channel { .. } => 8,
279            },
280            options: match self {
281                Self::String { options } => Some(options),
282                _ => None,
283            },
284            channel_types: match self {
285                Self::Channel { channel_types, default_channels: _ } => channel_types.as_deref(),
286                _ => None,
287            },
288            default_values: &default_values,
289        };
290
291        json.serialize(serializer)
292    }
293}
