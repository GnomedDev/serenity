8/// A builder to create or edit a [`Role`] for use via a number of model methods.
9///
10/// These are:
11///
12/// - [`GuildId::create_role`]
13/// - [`GuildId::edit_role`]
14/// - [`Role::edit`]
15///
16/// Defaults are provided for each parameter on role creation.
17///
18/// # Examples
19///
20/// Create a hoisted, mentionable role named `"a test role"`:
21///
22/// ```rust,no_run
23/// # use serenity::builder::EditRole;
24/// # use serenity::http::Http;
25/// # use serenity::model::id::GuildId;
26/// # use std::sync::Arc;
27/// #
28/// # async fn run() -> Result<(), Box<dyn std::error::Error>> {
29/// # let http: Arc<Http> = unimplemented!();
30/// # let guild_id: GuildId = unimplemented!();
31/// #
32/// // assuming a `guild_id` has been bound
33/// let builder = EditRole::new().name("a test role").hoist(true).mentionable(true);
34/// let role = guild_id.create_role(&http, builder).await?;
35/// # Ok(())
36/// # }
37/// ```
38///
39/// [Discord docs](https://discord.com/developers/docs/resources/guild#modify-guild-role)
40#[derive(Clone, Debug, Default, Serialize)]
41#[must_use]
42pub struct EditRole<'a> {
43    #[serde(skip_serializing_if = "Option::is_none")]
44    name: Option<Cow<'a, str>>,
45    #[serde(skip_serializing_if = "Option::is_none")]
46    permissions: Option<u64>,
47    #[serde(skip_serializing_if = "Option::is_none")]
48    #[serde(rename = "color")]
49    colour: Option<Colour>,
50    #[serde(skip_serializing_if = "Option::is_none")]
51    hoist: Option<bool>,
52    #[serde(skip_serializing_if = "Option::is_none")]
53    icon: Option<Option<Cow<'a, str>>>,
54    #[serde(skip_serializing_if = "Option::is_none")]
55    unicode_emoji: Option<Option<Cow<'a, str>>>,
56
57    #[serde(skip_serializing_if = "Option::is_none")]
58    mentionable: Option<bool>,
59
60    #[serde(skip)]
61    position: Option<i16>,
62    #[serde(skip)]
63    audit_log_reason: Option<&'a str>,
64}
65
66impl<'a> EditRole<'a> {
67    /// Equivalent to [`Self::default`].
68    pub fn new() -> Self {
69        Self::default()
70    }
71
72    /// Creates a new builder with the values of the given [`Role`].
73    pub fn from_role(role: &'a Role) -> Self {
74        EditRole {
75            hoist: Some(role.hoist()),
76            mentionable: Some(role.mentionable()),
77            name: Some(Cow::Borrowed(&role.name)),
78            permissions: Some(role.permissions.bits()),
79            position: Some(role.position),
80            colour: Some(role.colour),
81            unicode_emoji: role.unicode_emoji.as_ref().map(|v| Some(Cow::Borrowed(v.as_str()))),
82            audit_log_reason: None,
83            // TODO: Do we want to download role.icon?
84            icon: None,
85        }
86    }
87
88    /// Set the colour of the role.
89    pub fn colour(mut self, colour: impl Into<Colour>) -> Self {
90        self.colour = Some(colour.into());
91        self
92    }
93
94    /// Whether or not to hoist the role above lower-positioned roles in the user list.
95    pub fn hoist(mut self, hoist: bool) -> Self {
96        self.hoist = Some(hoist);
97        self
98    }
99
100    /// Whether or not to make the role mentionable, upon which users with that role will be
101    /// notified.
102    pub fn mentionable(mut self, mentionable: bool) -> Self {
103        self.mentionable = Some(mentionable);
104        self
105    }
106
107    /// Set the role's name.
108    pub fn name(mut self, name: impl Into<Cow<'a, str>>) -> Self {
109        self.name = Some(name.into());
110        self
111    }
112
113    /// Set the role's permissions.
114    pub fn permissions(mut self, permissions: Permissions) -> Self {
115        self.permissions = Some(permissions.bits());
116        self
117    }
118
119    /// Set the role's position in the role list. This correlates to the role's position in the
120    /// user list.
121    pub fn position(mut self, position: i16) -> Self {
122        self.position = Some(position);
123        self
124    }
125
126    /// Set the role icon to a unicode emoji.
127    pub fn unicode_emoji(mut self, unicode_emoji: Option<String>) -> Self {
128        self.unicode_emoji = Some(unicode_emoji.map(Into::into));
129        self.icon = Some(None);
130        self
131    }
132
133    /// Set the role icon to a custom image.
134    pub fn icon(mut self, icon: Option<&CreateAttachment<'_>>) -> Self {
135        self.icon = Some(icon.map(CreateAttachment::to_base64).map(Into::into));
136        self.unicode_emoji = Some(None);
137        self
138    }
139
140    /// Sets the request's audit log reason.
141    pub fn audit_log_reason(mut self, reason: &'a str) -> Self {
142        self.audit_log_reason = Some(reason);
143        self
144    }
145
146    /// Edits the role.
147    ///
148    /// **Note**: Requires the [Manage Roles] permission.
149    ///
150    /// # Errors
151    ///
152    /// Returns [`Error::Http`] if the current user lacks permission or if invalid data is given.
153    ///
154    /// [Manage Roles]: Permissions::MANAGE_ROLES
155    #[cfg(feature = "http")]
156    pub async fn execute(
157        self,
158        http: &Http,
159        guild_id: GuildId,
160        role_id: Option<RoleId>,
161    ) -> Result<Role> {
162        let role = match role_id {
163            Some(role_id) => {
164                http.edit_role(guild_id, role_id, &self, self.audit_log_reason).await?
165            },
166            None => http.create_role(guild_id, &self, self.audit_log_reason).await?,
167        };
168
169        if let Some(position) = self.position {
170            guild_id
171                .edit_role_positions(http, [(role.id, position)], self.audit_log_reason)
172                .await?;
173        }
174        Ok(role)
175    }
176}
